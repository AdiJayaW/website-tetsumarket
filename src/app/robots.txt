import { MetadataRoute } from 'next';

export default function robots(): MetadataRoute.Robots {
  const baseUrl = 'https://tetsumarket.my.id';

  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: [
        '/api/',       // Mencegah Google meng-crawl endpoint API backend
        '/admin/',     // Mencegah Google mengindeks halaman admin (jika ada)
        '/_next/',    // Mencegah bot membuang crawl budget pada asset internal Next.js
      ],
    },
    sitemap: `${baseUrl}/sitemap.xml`,
  };
}
